name: Dependency Audit & SBOM

on:
  push:
    branches: [main]
  schedule:
    - cron: "0 6 * * 1"  # weekly, Monday 6am UTC
  workflow_dispatch: {}

jobs:
  audit:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout repo
        uses: actions/checkout@v4

      - name: Set up Python
        uses: actions/setup-python@v5
        with:
          python-version: "3.11"

      - name: Install project + audit tools
        run: |
          python -m pip install --upgrade pip
          pip install -r requirements.txt
          pip install pip-audit cyclonedx-bom

      # Known-vulnerability scan against your actual installed versions
      - name: Run pip-audit (vulnerability scan)
        run: pip-audit -r requirements.txt --format json --output audit-report.json
        continue-on-error: true  # don't block CI, but still surface findings

      # Full bill-of-materials: exact versions of every dependency
      - name: Generate SBOM (CycloneDX)
        run: cyclonedx-py requirements -i requirements.txt -o sbom.json

      - name: Upload audit + SBOM artifacts
        uses: actions/upload-artifact@v4
        with:
          name: dependency-reports
          path: |
            audit-report.json
            sbom.json
          retention-days: 90

      - name: Fail build on HIGH/CRITICAL vulnerabilities
        run: |
          python - <<'EOF'
          import json, sys
          data = json.load(open("audit-report.json"))
          severe = [d for d in data.get("dependencies", []) if d.get("vulns")]
          if severe:
              print(f"Found {len(severe)} dependencies with known vulnerabilities:")
              for d in severe:
                  print(f"  - {d['name']} {d['version']}")
              sys.exit(1)
          EOF
