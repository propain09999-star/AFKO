name: Required Checks (Orchestrator)

on:
  push:
    branches: [main]
  pull_request:
    branches: [main]

jobs:
  # --- Detect what actually changed, so we don't burn CI minutes
  #     re-running Docker/Ollama-heavy jobs for e.g. a README edit ---
  changes:
    runs-on: ubuntu-latest
    outputs:
      code: ${{ steps.filter.outputs.code }}
      docker: ${{ steps.filter.outputs.docker }}
      deps: ${{ steps.filter.outputs.deps }}
    steps:
      - uses: actions/checkout@v4
      - uses: dorny/paths-filter@v3
        id: filter
        with:
          filters: |
            code:
              - '**/*.py'
              - 'tests/**'
            docker:
              - 'Dockerfile'
              - 'docker-compose*.yml'
            deps:
              - 'requirements*.txt'

  # --- Priority 1: build/test ---
  ci:
    needs: changes
    if: needs.changes.outputs.code == 'true' || needs.changes.outputs.deps == 'true'
    uses: ./.github/workflows/ci.yml

  # --- Priority 2: security (known + unknown CVEs, secrets) ---
  security:
    needs: changes
    if: needs.changes.outputs.code == 'true' || needs.changes.outputs.deps == 'true' || needs.changes.outputs.docker == 'true'
    uses: ./.github/workflows/security-deep-scan.yml

  # --- Priority 3: deterministic logic-bug testing ---
  deterministic-tests:
    needs: [changes, ci]
    if: needs.changes.outputs.code == 'true'
    uses: ./.github/workflows/deterministic-bug-testing.yml

  # --- Single aggregated status: THIS is the one check you mark
  #     "required" in branch protection, instead of picking 3 separately ---
  required-status:
    runs-on: ubuntu-latest
    needs: [ci, security, deterministic-tests]
    if: always()
    steps:
      - name: Check all required jobs passed
        run: |
          if [ "${{ needs.ci.result }}" = "failure" ] || \
             [ "${{ needs.security.result }}" = "failure" ] || \
             [ "${{ needs.deterministic-tests.result }}" = "failure" ]; then
            echo "One or more required checks failed."
            exit 1
          fi
          echo "All required checks passed (or were skipped as not applicable)."
