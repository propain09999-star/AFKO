name: Performance Benchmark

on:
  push:
    branches: [main]
  pull_request:
    branches: [main]

jobs:
  benchmark:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: "3.11"

      - name: Install dependencies
        run: |
          python -m pip install --upgrade pip
          pip install -r requirements.txt
          pip install pytest-benchmark

      # NOTE: expects a small fixed test clip checked into
      # tests/fixtures/sample_broadcast.wav — a short recording
      # with a known keyword mention, used as the constant baseline
      - name: Benchmark transcription speed
        run: |
          pytest tests/benchmark_transcribe.py \
            --benchmark-only \
            --benchmark-json=bench-transcribe.json

      - name: Benchmark detection accuracy (known keyword hit)
        run: python tests/benchmark_detect_accuracy.py --fixture tests/fixtures/sample_broadcast.wav

      - name: Compare against baseline
        uses: benchmark-action/github-action-benchmark@v1
        with:
          tool: "pytest"
          output-file-path: bench-transcribe.json
          github-token: ${{ secrets.GITHUB_TOKEN }}
          auto-push: true
          alert-threshold: "150%"  # flag if a change makes it 1.5x slower
          comment-on-alert: true
          fail-on-alert: true
