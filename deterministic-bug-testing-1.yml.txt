name: Deterministic Bug Testing (Property-Based + Coverage)

on:
  push:
    branches: [main]
  pull_request:
    branches: [main]
  workflow_call: {}

jobs:
  property-based-testing:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: "3.11"

      - name: Install dependencies
        run: |
          python -m pip install --upgrade pip
          pip install -r requirements.txt
          pip install -r requirements-dev.txt
          pip install hypothesis pytest-cov

      # Hypothesis generates hundreds of input variations per test
      # (e.g. malformed audio chunks, edge-case keyword strings,
      # truncated config.yaml fields) and checks invariants hold —
      # this is what surfaces logic bugs your fixed test cases miss,
      # and unlike the flaky-detection workflow, every failure here
      # is 100% reproducible on the exact input that broke it.
      - name: Run property-based tests
        run: pytest tests/ -m hypothesis -v --hypothesis-seed=0

      - name: Run full suite with coverage
        run: pytest tests/ --cov=. --cov-report=xml --cov-report=term

      # Fail the build if coverage drops below a floor — an
      # under-tested code path is where deterministic bugs hide
      - name: Enforce coverage floor
        run: |
          python -m coverage report --fail-under=70

      - name: Upload coverage report
        uses: actions/upload-artifact@v4
        with:
          name: coverage-report
          path: coverage.xml
